const productos= [
    {id: 1, nombre: 'Mouse', precio: 25},
    {id: 2, nombre: 'Teclado', precio: 45},
    {id: 3, nombre: 'Monitor', precio: 150},
    {id: 4, nombre: 'Laptop', precio: 800}

productos.push({id: 4, nombre: 'Parlante', precio: 30});

//filter: se queda con todos los que cumplen la condicion
const sinMouse = productos.filter((p) => p.id !== 1);
